$\text{WE NEED TO INTRODUCE POLAR CORRDINATE FIRST :}$  
>$\text{Using Polar corrdinate :}$  
>$$
    \begin{cases}
        x =r\cos(\theta) \\

>$\text{Then , }H(r\cos(\theta),r\sin(\theta))=H(g(r,\theta))=H \circ g(r,\theta)$  
>$\color{green}{\text{Denote }\tilde{H} \coloneqq H \circ g}$  
>$\text{Now , we have one Definition and one Theroem to introduce to you:}$  
>$\color{orange}{\text{Definition :}(balanced \; point)}$  
>$\color{orange}{\text{Given a Hamiltonian fucntion }H(x,y) \text{ , we say that a point }(\tilde{x},\tilde{y}) \text{is a }balanced \; point \text{ of }H\text{ if :}}$  
>$$\color{orange}{\begin{cases}
        H_x(\tilde{x},\tilde{y})=0  \\
        H_y(\tilde{x},\tilde{y})=0
        \end{cases}}
>$$  
>$\color{red}{\text{Theorem :}}$  
>$\color{red}{\text{Given a Hamiltonian function }H(x,y) ,\text{ and a Polar corrdinate as :}}$  
>$$
\color{red}{ \begin{cases}
        x =r\cos(\theta) \\
        y =r\sin(\theta)
    \end{cases}}
>$$  
>$\color{red}{\text{Then :}}$  
>$$
    \color{red}{(\tilde{x},\tilde{y}) \text{ is a balanced point of }H \iff (\tilde{r},\tilde{\theta}) \text{ is a balanced point of } \tilde{H}}
>$$  
>$\color{red}{\text{where :}}$  
>$$
\color{red}{ \begin{cases}
        \tilde{x} =\tilde{r}\cos(\tilde{\theta}) \\
        \tilde{y} =\tilde{r}\sin(\tilde{\theta})
    \end{cases}}
>$$  
>$\color{blue}{\text{PROOF OF THE THEROREM :}}$  
>$\color{blue}{\text{On one hand:}}$  
>$$
\color{blue}{\mathscr{D}_{\tilde{H}(r,\theta)}=\begin{bmatrix}
    \frac{\partial \tilde{H}}{\partial r} & \frac{\partial \tilde{H}}{\partial \theta}
\end{bmatrix}}
>$$  
>$\color{blue}{\text{On the other hand:}}$  
>$$
\color{blue}{\mathscr{D}_{\tilde{H}(\tilde{r},\tilde{\theta})} \overset{\text{chain rule in inf3}}{=} \mathscr{D}_{H(g(r,\theta))} \circ \mathscr{D}_{g(r,\theta)}}
>$$  
>$$
\color{blue}{= \begin{bmatrix}
    \frac{\partial H}{\partial x}(r\cos(\theta),r\sin(\theta)) & \frac{\partial H}{\partial y}(r\cos(\theta),r\sin(\theta))
\end{bmatrix} \circ \begin{bmatrix}
    \cos(\theta) & -r\sin(\theta) \\ 
    \sin(\theta) & r\cos(\theta)
\end{bmatrix} } 
>$$  
>$\color{blue}{\text{Therefore :}}$  
>$$
\color{blue}{\begin{bmatrix}
    \frac{\partial \tilde{H}}{\partial r} & \frac{\partial \tilde{H}}{\partial \theta}
\end{bmatrix} = \begin{bmatrix}
    \frac{\partial H}{\partial x}(r\cos(\theta),r\sin(\theta)) & \frac{\partial H}{\partial y}(r\cos(\theta),r\sin(\theta))
\end{bmatrix} \circ \begin{bmatrix}
    \cos(\theta) & -r\sin(\theta) \\ 
    \sin(\theta) & r\cos(\theta)
\end{bmatrix}} \tag{*}
>$$
>$\color{red}{(\Rightarrow):}$  
>$\color{blue}{\text{Suppose }(\tilde{x},\tilde{y}) \text{ is a balanced point of }H }$  
>$\color{blue}{\text{Then by definition :}}$  
>$$
\color{blue}{\begin{cases}
        H_x(\tilde{x},\tilde{y})=0  \\
        H_y(\tilde{x},\tilde{y})=0
        \end{cases}}
>$$  
>$\color{blue}{\text{By (*) implies that :}}$  
>$$
    \color{blue}{\begin{cases}
        \frac{\partial \tilde{H}}{\partial r}(\tilde{r},\tilde{\theta}) = 0 \cdot \cos(\tilde{\theta}) + 0 \cdot \sin(\tilde{\theta}) =0 \\ 
        \frac{\partial \tilde{H}}{\partial \theta}(\tilde{r},\tilde{\theta}) = 0 \cdot (-\tilde{r}\sin(\tilde{\theta})) + 0 \cdot (\tilde{r}\cos(\tilde{\theta})) =0
    \end{cases} }
>$$  
>$\color{red}{\text{Therefore , }(\tilde{r},\tilde{\theta}) \text{ is a balanced point of } \tilde{H}}$  
>$\color{red}{(\Leftarrow):}$  
>$\color{blue}{\text{Suppose }(\tilde{r},\tilde{\theta}) \text{ is a balanced point of } \tilde{H}}$  
>$\color{blue}{\text{Then by definition :}}$  
>$$\color{blue}{\begin{cases}
        \tilde{H}_r(\tilde{r},\tilde{\theta})=0  \\
        H_y(\tilde{r},\tilde{\theta})=0
        \end{cases}}
>$$  
>$\color{blue}{\text{In (*) , since the matirx :}}$  
>$$
\color{blue}{\begin{bmatrix}
    \cos(\theta) & -r\sin(\theta) \\ 
    \sin(\theta) & r\cos(\theta)
\end{bmatrix}}
>$$  
>$\color{blue}{\text{is invertible}(det = r >0)}$  
>$\color{blue}{\text{So,(*) implies that :}}$  
>$$
\color{blue}{\begin{bmatrix}
    \frac{\partial H}{\partial x}(r\cos(\theta),r\sin(\theta)) & \frac{\partial H}{\partial y}(r\cos(\theta),r\sin(\theta))
\end{bmatrix} = \begin{bmatrix}
    0 & 0
\end{bmatrix}  }
>$$  
>$\color{blue}{\text{Therefore : }}$  
>$$
\color{blue}{\begin{cases}
        H_x(\tilde{x},\tilde{y})=0  \\
        H_y(\tilde{x},\tilde{y})=0
        \end{cases}}
>$$  
>$\color{red}{\text{Therefore , }(\tilde{x},\tilde{y}) \text{ is a balanced point of }}H$  
>$\color{blue}{\text{THEOREM IS DONE.}}$  
>$\color{blue}{\blacksquare}$

- $\color{blue}{\text{CASE }\mathrm{II} \text{ : }B_2=0}$  
$\text{Then :} \begin{cases}
    h(\theta) = -A_1\sin(\theta) +\frac{B_1}{3}\sin(3\theta)    \\
    h'(\theta) = -A_1\cos(\theta) +\frac{B_1}{3}\cos(3\theta) \overset{\text{Simplify}}{=} \cos(\theta)(4B_1\cos^2(\theta)-(A_1+3B_1))

$\text{So , }h'(\tilde{\theta})=0 \iff \cos(\tilde{\theta})=0 \text{ or } (4B_1\cos^2(\tilde{\theta})-(A_1+3B_1))=0$  
$\color{green}{\text{If }\cos(\tilde{\theta})=0 :}$

$$

$$

 0 \le \frac{A_1+3B_1}{4B_1} \le 1

$\Rightarrow \cos^2(\tilde{\theta})=1 \Rightarrow \tilde{\theta} = 0 \text{ or } \pi , \text{ But }h(0)=h(\pi)=0 \not> 0 , \text{ impossible !}$  
$\text{So , }\frac{A_1+3B_1}{4B_1} < 1$  
$\text{Therefore , }$  
$$
0<\frac{A_1+3B_1}{4B_1}<1
$$    
$\text{Now , let }t \coloneqq \cos(\tilde{\theta})$  
$\text{Then :} t^2 = \frac{A_1+3B_1}{4B_1}$  
$\text{So , }$  
$h(\tilde{\theta}) = -A_1\sin(\tilde{\theta})+\frac{B_1}{3}\sin(3\tilde{\theta})$  
$=-A_1\sin(\tilde{\theta})+\frac{B_1}{3}(3\sin(\tilde{\theta})-4\sin^3(\tilde{\theta}))$  
$=-A_1\sin(\tilde{\theta})+\frac{B_1}{3}(\sin(\tilde{\theta})(4\cos^2(\tilde{\theta})-1))$  
$=-A_1\sin(\tilde{\theta})+\frac{B_1}{3}(\sin(\tilde{\theta})(4t^2-1))$  
$=\sin(\tilde{\theta})(-A_1+\frac{B_1}{3}(4t^2-1))$  
$=\sin(\tilde{\theta})(-A_1+\frac{B_1}{3}(\frac{A_1+2B_1}{B_1}))$  
$=\sin(\tilde{\theta})(-A_1+\frac{A_1+2B_1}{3})$  
$=\frac{2}{3}(B_1-A_1)\sin(\tilde{\theta})$  
$\text{Now , consider the equation :}$  
$$
\cos^2(\tilde{\theta})=\frac{A_1+3B_1}{4B_1}
$$  
$\tilde{\theta} \text{ has 4 solutions in }[0,\pi) \text{ , 2 of them has }\sin >0 \text{ , 2 of them has } \sin <0$  
$\text{Thus , } \begin{cases}
    \text{If }(B_1-A_1)>0 \text{ choose the first 2 of them.}   \\
    \text{If }(B_1-A_1)<0 \text{ choose the second 2 of them.}
\end{cases}$  
$\color{red}{\text{It tell us that : }}$  
$$
\color{red}{0<\frac{A_1+3B_1}{4B_1}<1 \Rightarrow \text{(**) always has solutions !}}
$$
$\text{To simplify it :}$  
$$
  0<\frac{A_1+3B_1}{4B_1}<1 \iff (A_1+3B_1)(A_1-B_1)<0
$$
$\text{Therefore , conclude two CASES. The bifurcation diagram of }\Sigma_2 \text{ is :}$  
$$
    \color{red}{A_1=0 \text{ or } (B_2=0,(A_1-B_1)(A_1+3B_1)<0)}
$$
